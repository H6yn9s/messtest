import { createClient } from "https://cdn.jsdelivr.net/npm/@supabase/supabase-js/+esm"; const SUPABASE_URL = "PASTE_SUPABASE_URL"; const SUPABASE_ANON = "PASTE_SUPABASE_ANON_KEY"; const supabase = createClient(SUPABASE_URL, SUPABASE_ANON); // ---------- helpers ---------- const $ = (id) => document.getElementById(id); function setHint(text){ const h = $("hint"); if(h) h.textContent = text || ""; } function normalizeUsername(u){ return (u || "") .trim() .toLowerCase() .replace(/[^a-z0-9_]/g, ""); } function isProfilePage(){ return location.pathname.includes("profile.html") || location.pathname.includes("/u/"); } function getUsernameFromPath(){ // поддержка: // /u/username // /profile.html?u=username const p = location.pathname; const idx = p.indexOf("/u/"); if(idx !== -1){ return p.substring(idx + 3).split("/")[0]; } const url = new URL(location.href); return url.searchParams.get("u"); } function applyTheme(profile){ const accent = profile?.accent || "#7c3aed"; document.documentElement.style.setProperty("--accent", accent); const bg = $("bg"); if(!bg) return; if(profile?.theme === "flat"){ bg.style.background = "linear-gradient(180deg, #0a0a0f, #07070b)"; } else { bg.style.background = `radial-gradient(circle at 30% 20%, ${accent}33, transparent 45%), radial-gradient(circle at 80% 70%, rgba(0,200,255,0.14), transparent 50%), linear-gradient(180deg, #0a0a0f, #07070b)`; } } function goToUser(username){ // GitHub Pages не умеет нормальные роуты без 404.html // поэтому делаем fallback: // profile.html?u=username location.href = `./profile.html?u=${encodeURIComponent(username)}`; } // ---------- index page ---------- async function initIndex(){ const tabLogin = $("tabLogin"); const tabRegister = $("tabRegister"); const loginBox = $("loginBox"); const registerBox = $("registerBox"); if(!tabLogin) return; tabLogin.onclick = () => { tabLogin.classList.add("active"); tabRegister.classList.remove("active"); loginBox.style.display = "block"; registerBox.style.display = "none"; setHint(""); }; tabRegister.onclick = () => { tabRegister.classList.add("active"); tabLogin.classList.remove("active"); registerBox.style.display = "block"; loginBox.style.display = "none"; setHint(""); }; $("btnLogin").onclick = async () => { setHint("Входим..."); const email = $("loginEmail").value.trim(); const password = $("loginPassword").value; const { data, error } = await supabase.auth.signInWithPassword({ email, password }); if(error){ setHint("Ошибка: " + error.message); return; } const user = data.user; const { data: prof } = await supabase .from("profiles") .select("username") .eq("id", user.id) .maybeSingle(); if(!prof?.username){ setHint("Профиль не найден. Создай новый аккаунт."); return; } setHint("Готово!"); goToUser(prof.username); }; $("btnRegister").onclick = async () => { setHint("Создаём аккаунт..."); let username = normalizeUsername($("regUsername").value); const email = $("regEmail").value.trim(); const password = $("regPassword").value; if(username.length < 3){ setHint("Username минимум 3 символа (a-z 0-9 _)."); return; } const { data: exists } = await supabase .from("profiles") .select("username") .eq("username", username) .maybeSingle(); if(exists?.username){ setHint("Этот username занят."); return; } const { data, error } = await supabase.auth.signUp({ email, password }); if(error){ setHint("Ошибка: " + error.message); return; } const user = data.user; if(!user){ setHint("Проверь email (возможно нужна верификация)."); return; } const { error: insertErr } = await supabase .from("profiles") .insert({ id: user.id, username, display_name: username, bio: "", avatar_url: "", theme: "gradient", accent: "#7c3aed" }); if(insertErr){ setHint("Ошибка создания профиля: " + insertErr.message); return; } setHint("Аккаунт создан!"); goToUser(username); }; } // ---------- profile page ---------- async function uploadAvatar(file, userId){ const ext = file.name.split(".").pop(); const path = `${userId}/${Date.now()}.${ext}`; const { error } = await supabase.storage .from("avatars") .upload(path, file, { upsert: true }); if(error) throw error; const { data } = supabase.storage.from("avatars").getPublicUrl(path); return data.publicUrl; } function renderLinks(list){ const wrap = $("links"); if(!wrap) return; wrap.innerHTML = ""; for(const l of list){ const a = document.createElement("a"); a.className = "linkBtn"; a.href = l.url; a.target = "_blank"; a.rel = "noopener noreferrer"; a.textContent = l.title; wrap.appendChild(a); } } function renderLinksEditor(list, userId){ const wrap = $("linksEditor"); if(!wrap) return; wrap.innerHTML = ""; for(const l of list){ const row = document.createElement("div"); row.className = "linkItem"; const t = document.createElement("input"); t.className = "input"; t.value = l.title; const u = document.createElement("input"); u.className = "input"; u.value = l.url; const del = document.createElement("button"); del.textContent = "Удалить"; del.onclick = async () => { await supabase.from("links").delete().eq("id", l.id).eq("user_id", userId); await loadProfilePage(); }; t.onchange = async () => { await supabase.from("links").update({ title: t.value }).eq("id", l.id).eq("user_id", userId); }; u.onchange = async () => { await supabase.from("links").update({ url: u.value }).eq("id", l.id).eq("user_id", userId); }; row.appendChild(t); row.appendChild(u); row.appendChild(del); wrap.appendChild(row); } } async function loadProfilePage(){ const username = normalizeUsername(getUsernameFromPath()); if(!username){ document.body.innerHTML = "
Нет username
"; return; } // профиль const { data: profile, error } = await supabase .from("profiles") .select("*") .eq("username", username) .maybeSingle(); if(error || !profile){ document.body.innerHTML = "
Профиль не найден
"; return; } applyTheme(profile); $("displayName").textContent = profile.display_name || profile.username; $("bio").textContent = profile.bio || ""; const ava = profile.avatar_url || "https://api.dicebear.com/7.x/thumbs/svg?seed=" + profile.username; $("avatar").src = ava; // ссылки const { data: links } = await supabase .from("links") .select("*") .eq("user_id", profile.id) .order("sort", { ascending: true }) .order("id", { ascending: true }); renderLinks(links || []); // проверяем авторизацию const { data: auth } = await supabase.auth.getUser(); const me = auth?.user; const editPanel = $("editPanel"); if(!editPanel) return; if(me && me.id === profile.id){ editPanel.style.display = "block"; $("editName").value = profile.display_name || ""; $("editUsername").value = profile.username || ""; $("editBio").value = profile.bio || ""; $("editTheme").value = profile.theme || "gradient"; $("editAccent").value = profile.accent || "#7c3aed"; $("btnSave").onclick = async () => { const newUsername = normalizeUsername($("editUsername").value); if(newUsername.length < 3){ alert("Username минимум 3 символа"); return; } // если username меняется — проверяем занятость if(newUsername !== profile.username){ const { data: exists } = await supabase .from("profiles") .select("username") .eq("username", newUsername) .maybeSingle(); if(exists?.username){ alert("Username занят"); return; } } let avatarUrl = profile.avatar_url; const file = $("avatarFile").files?.[0]; if(file){ try{ avatarUrl = await uploadAvatar(file, profile.id); }catch(e){ alert("Ошибка загрузки аватара: " + e.message); return; } } const payload = { display_name: $("editName").value.trim(), username: newUsername, bio: $("editBio").value.trim(), theme: $("editTheme").value, accent: $("editAccent").value, avatar_url: avatarUrl, updated_at: new Date().toISOString() }; const { error: updErr } = await supabase .from("profiles") .update(payload) .eq("id", profile.id); if(updErr){ alert("Ошибка сохранения: " + updErr.message); return; } // если username поменяли — перейти на новый if(newUsername !== username){ goToUser(newUsername); return; } await loadProfilePage(); }; $("btnLogout").onclick = async () => { await supabase.auth.signOut(); location.href = "./index.html"; }; $("btnAddLink").onclick = async () => { const title = $("linkTitle").value.trim(); const url = $("linkUrl").value.trim(); if(!title || !url){ alert("Заполни title и url"); return; } await supabase.from("links").insert({ user_id: profile.id, title, url, sort: 0 }); $("linkTitle").value = ""; $("linkUrl").value = ""; await loadProfilePage(); }; renderLinksEditor(links || [], profile.id); } else { editPanel.style.display = "none"; } } // ---------- init ---------- if(location.pathname.endsWith("index.html") || location.pathname === "/" || location.pathname.endsWith("/")){ initIndex(); } else { // profile loadProfilePage(); }
